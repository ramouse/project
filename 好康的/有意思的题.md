隔两个数交换

因为数组单调递增，想保证一个数要比旁边两个数都小或都大，如果每两个都交换，会使交换次数变多且不好找到规律

因为数组单调递增，把一个较小的数换到两个肯定比他大的数之间，就能完成题目要求的排序

```c++
#include<bits/stdc++.h>
using namespace std;

int main(){
    int n;
    cin>>n;
    vector<int> a(n);
    for(int i = 0;i<n;i++){
        a[i] = i +1;
    }
    
    for(int i = 0;i + 1<n;i+=2){
        swap(a[i],a[i+1]);
    }
    
    for(auto i : a){
        cout<<i<<" ";
    }
    
    return 0;
}
```





##超大数据量的判断子字符串

[E-烤馍片_华中农业大学第十四届程序设计竞赛（新生赛）同步赛](https://ac.nowcoder.com/acm/contest/95480/E)

此题非常容易想到双指针判断，想法没错！但是，由于数据量异常之大，可能会导致时间炸掉或者输出错误！

所以我们可以按块统计，不必展开字符串

```c++
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

int main()
{
    int n;
    cin >> n;
    vector<pair<ll,char>> blocks;
    while (n--)
    {
        ll a;
        char ch;
        cin >> a >> ch;
        blocks.push_back({a, ch});
    }

    int m;
    cin >> m;
    string t;
    cin >> t;

    ll ans = 0;
    if (m == 1) {
        char need = t[0];
        for (auto &bk : blocks) if (bk.second == need) ans += bk.first;
    } else { // m == 2
        if (t[0] == t[1]) {
            for (auto &bk : blocks) if (bk.second == t[0] && bk.first >= 2) ans += (bk.first - 1);
        } else {
            for (ll i = 0; i + 1 < blocks.size(); ++i)
                if (blocks[i].second == t[0] && blocks[i+1].second == t[1]) ans += 1;
        }
    }

    cout << ans;
    return 0;
}

```
